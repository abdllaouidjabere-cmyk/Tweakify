def optimize_performance(self):
    """Applies all performance tweaks"""
    results = []
    
    # Disable visual animations
    ok = set_registry(
        HKCU, "Control Panel\\Desktop",
        "MinAnimate", 0
    )
    results.append(("Animations", ok))

    # Set High Performance power plan
    run_cmd(
        "powercfg /setactive 8c5e7fda"
    )
    
    return results
